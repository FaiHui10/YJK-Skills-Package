public class Hi_API_YJKDTReport
{
public:
Hi_API_YJKDTReport();
~Hi_API_YJKDTReport();
//	模型准备
//  projPath:工程路径
//  modelType：0 - 主模型 1 - 中震 2 - 大震
void init(String projPath, int modelType);
//	1
//	获取工况信息
//	vLoadCaseComb 地震波（含工况）列表
//	vLoadCase  地震波列表
//	vLoadComb  工况组合列表
void getVLoadComb(List<String> vLoadCaseComb, List<String> vLoadCase, List<String> vLoadComb);
//	2
//	获取地震波角度
BOOL getVLoadAngle(Dictionary<String, String> vLoadCaseAngle);
//	3
//	获取地震波对应时刻表
BOOL getVLTimeTbale(Dictionary<String, List<double>> vLoadTimeTable);
//	4
//	获取节点时程数据
//	0-位移	1-速度	2-加速度
//	vector[i] 第一层为工况列
//	vector[i][j] 第二层为方向  0-X 1-Y 2-Z 3-Mx 4-My 5-Mz
//	vector[i][j][k] 第三层为时程数据
void getNodeDispVecAccHistoryRst(int iRstType, Dictionary<int, List<List<List<double>>>> m_NodeHisRst);
//	5
//	获取楼层最大响应
//	0-楼层位移    1-楼层位移角  2-楼层加速度  3-楼层质心绝对加速度
//	4-楼层剪力    5-楼层弯矩    6-楼层轴力
//	楼层结果包括主次向结果及CQC结果主次向结果
//	CQC结果存于最后一个工况，例共8个地震工况，则此处vector大小为9。
void getFlrDispForceMaxRst(int iRstType, Dictionary<Tuple<int, int>, Mdl_DTHFLOORINDEX> m_FlrMaxDispForceRst);
//	6
//	获取楼层响应时程数据
//	0-楼层位移    1-楼层位移角  2-楼层加速度  3-楼层质心绝对加速度
//	4-楼层剪力    5-楼层弯矩    6-楼层轴力
//	pair<int, int> -层，塔
//	vector[i] 第一层为主次列
//	vector[i][j] 第二层为工况列
//	vector[i][j][k] 第三层为时程数据
//	楼层结果包括主次向结果
void getFlrDispForceHistoryRst(int iRstType, Dictionary<Tuple<int, int>, List<List<List<double>>>> _m_FlrMaxDispForceRst);
//	7
//	获取结构所有工况的能量数据
//	vector[i] 第一层为工况列
//	vector[i][j] 第二层定义：
//	0-时间点    1-动能  2-阻尼耗能   3-内能
//	4-外力做功  5-速度型阻尼器耗能    6-位移型阻尼器耗能
//  7-隔震支座耗能
void getStruEnergyData(List<List<List<double>>> m_StruEnergyData);
//	8
//	获取所有连接单元的位移、速度、内力时程响应
//	键为连接单元ID
//	vector[i]          第一层为响应类型，0-位移  1-速度 2-内力
//	vector[i][j]       第二层为工况列表
//	vector[i][j][k]    第三层为自由度  0-x 1-y 2-z 3-Mx 4-My 5-Mz
//	vector[i][j][k][l] 第四层为各时程数据
void getAllLinkDispForceRst(Dictionary<int, List<List<List<List<double>>>>> _m_AllLinkDispForceRst);
//	9
//	获取模型的楼面水平加速度验算结果（减震）
//	所有工况下的楼层的最大加速度（主次方向的合加速度）
//	<层-塔>-合加速度值
void getFlrHrizontalAcc(Dictionary<Tuple<int, int>, double> m_FlrHriMaxAcc);
//	10
//	获取模型的层间位移角 （减震）
//	vector为各工况  层、塔-主、次位移角
void getFlrMaxAngle(List<Dictionary<Tuple<int, int>, List<double>>>  m_FlrMaxAngle);
//	11
//	是否是隔震结构（隔震）
BOOL IsIsoStructure();
//	12
//	获取隔震支座应力、位移、内力  输出每个工况下每个支座的响应值
//	vLoadCase存放用户选择的地震工况名称
//	vLoadComb存放用户选择的工况组合名称
//	0-CStress0, 1-CStressMax, 2-CStressMin, 3-TStressMax, 4-HDispMax
//	5-FxMax, 6-FxMin, 7-FyMax, 8-FyMin, 9-FzMax, 10-FzMin
BOOL getIsoStressDispForce(List<String> vLoadCase, List<String> vLoadComb, List<Dictionary<int, List<double>>> m_IsoStressDispForce);
//	13
//	获取模型抗倾覆验算结果（隔震）
//	层-塔-工况-主次值
//	输出全部楼层的抗倾覆验算结果，但抗倾覆验算报告中应只取隔震层及以上
//	m_antiOverMoment为抗倾覆弯矩  m_OverMoment为倾覆弯矩 m_safeCoe为安全系数
void getFlrOverturningRst(Dictionary<Tuple<int, int>, List<List<double>>>  m_antiOverMoment, Dictionary<Tuple<int, int>, List<List<double>>>  m_OverMoment,
Dictionary<Tuple<int, int>, List<List<double>>> m_safeCoe);
//	14
//	获取模型恢复力验算结果（隔震）
//	恢复力验算结果仅输出隔震层数据
//	vIsoFloorForce为恢复力  vIsoFloorYieldForce为屈服力
void getFlrRecoverForceRst(Dictionary<Tuple<int, int>, List<List<double>>> vIsoFloorForce, Dictionary<Tuple<int, int>, List<List<double>>> vIsoFloorYieldForce);
//	15
//	获取各阻尼器耗能及总应变能、阻尼耗能、附加阻尼比信息   --不需处理
//	vDampEnergy 键为阻尼器ID，值vector为各工况的阻尼耗能
//	vAllStruDampEnergyAndRatio为3*n数组，依次存放各工况的总应变能、阻尼耗能、附加阻尼比
//	IsAddNew 是否添加能量比值法附加阻尼比结果，若添加 vAllStruDampEnergyAndRatio 变为7*n，
//			 后四项为能量比值法计算得到的:速度型阻尼器耗能；位移形阻尼器耗能；阻尼耗能；附加阻尼比
void getDampAndEnergyAddRato(Dictionary<int, List<double>> vDampEnergy, List<List<double>> vAllStruDampEnergyAndRatio, int IsAddNew);
//	16
//	获取结构相对响应结果
//	0-楼层相对位移
//	MaxRelRstFloorTow <层，塔>,vector<vector<double>>  第一层按工况排列，第二层存放主方向和次方向的楼层相对响应
BOOL getFlrRelativeRst(int iRstType, Dictionary<Tuple<int, int>, List<List<double>>> MaxRelRstFloorTow);
//	17
//	获得反应谱曲线
//	图片存放位置在   工程目录//DYNAMIC//反应谱曲线.jpg
BOOL getSpecGraph(String Emf_Path, ref int nWidth, ref int nHeight);
//	18
//	获取所有框架单元的内力时程结果
//	map<工程名-pair<frameID,内力时程>>
//	0-5 I端   6-11 J端 依次为F1 F2 F3 M1 M2 M3
bool GetAllFrameHisForceRst(Dictionary<String, List<Tuple<int, List<List<double>>>>> vForce);
//	19
//	获取所有工况位移云图
bool GetAllLoadDisCloudGraph(Dictionary<Tuple<String, String>, List<Tuple<String, Tuple<int, int>>>> vGraphPath);
//	20
//	获取连接单元滞回曲线图
bool GetLinkForceDisGraph(Dictionary<String, List<Tuple<Tuple<int, int>, String>>> vCycleGrap);
//	21
//	获取楼层剪力、位移（包含隔震层位移）、应变能等信息
//	pair<int, int> -<层，塔>
//	vector<vector<vector<double>>> 第一层对应工况列表
//	vector<vector<double>> 0-剪力  1-位移  2-应变能
//	vector<double> 0-主方向结果  1-次方向结果
bool GetFlrTowForceDisEng(Dictionary<Tuple<int, int>, List<List<List<double>>>> vFlrTowData);
//	22
//	获取各角度地震波主方向剪力、CQC剪力等信息  <wdyna.out>
//	pair<float, CString>   -角度，地震波名
//	vector<pair<pair<int, int>, vector<double>>>  存放每个层塔的工况剪力和cqc剪力，最内层的pair<int,int> -<层,塔>
//	vector<double>  -存放两个值 0-工况主方向剪力，1-cqc主方向剪力 2-比值 3-放大系数
bool GetAngleLCForceInfo(Dictionary<Tuple<float, String>, List<Tuple<Tuple<int, int>, List<double>>>> vFlrTowForce);
//	23
//	获取弹性连接连接单元ID及类别
//  pair<int,CString> -int为连接单元ID， CString为连接单元类别
bool GetDynamicLinkInfo(List<Dictionary<int, String>> vLinkInfo);
//	24
//	获取弹性时程等效刚度、等效阻尼等
//	iModelType:0-小震  1-中震  2-大震 （对应弹性时程模型选择，小震 为 DYNAMIC文件夹， 中震 DYNAMIC_sf  大震 DYNAMIC_HY）
//
bool GetLinkEquKeCe(Dictionary<int, Mdl_LinkKeCeData> LinkKeCeRes, int iModelType);
}
