// 模型映射管理器
public ref class Hi_ModelMapping
{
private:
IModelMappingManager* m_native;
// 构造函数（内部使用，外部通过 Create 获取实例）
Hi_ModelMapping(IModelMappingManager* native);
public:
// 创建并获取模型映射管理器实例（单例）
static Hi_ModelMapping^ Create();
// 根据建模ID查找对应的全楼ID列表
List<int>^ GetTotalIDs(int modelID);
// 根据全楼ID和构件类型查找对应的建模ID列表，Mdl_AID_GJKind为构件类型枚举类
List<int>^ GetModelIDs(int totalID, Mdl_AID_GJKindgjKind);
// 根据墙的建模ID查找对应的全楼ID和类型列表
List<Tuple<int, int>^>^ GetWallTotalIDs(int modelID);
// 根据墙的全楼ID查找对应的建模ID和类型列表
List<Tuple<int, int>^>^ GetWallModelIDs(int totalID);
// 判断映射管理器是否为空
bool IsEmpty();
};

## 2. 转图相关接口说明

// 模型映射管理器
public ref class Hi_LayerDocEditor
{
public:
//全部「可读构件类型名」列表。
List<string> GetElementTypes();

bool GetElementLayerNames(string type, List<string> layers);
// 判断类型名是否在映射中存在（经内部规范化后匹配）。
bool IsElementTypeExists(string type);

Dictionary<string, List<string>> GetElementsLayerNames(List<string> types);

bool SetElementLayerNames(string type, List<string> layers, ref string failedNotFoundTypes);

bool SetElementLayerNames(Dictionary<string, List<string>> items, ref string failedNotFoundTypes);
};
