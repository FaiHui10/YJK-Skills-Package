//================================================
// 获取ID值如  int id1 = YPSC_GETWAVEPARA_ID(para)
#define YP_SCWAVEPARA_ID	0x0FFFFFFF	//用于取地震波参数中的ID (例 id = para & YP_SCWAVEPARA_ID)
#define YPSC_GETWAVEPARA_ID(para)  	\
((para) & YP_SCWAVEPARA_ID)
// 设置参数para的ID值为id
#define YPSC_SETWAVEPARA_ID(para, id) \
(para) = (para) & (~YP_SCWAVEPARA_ID) | (id);
// 获取PRIM值如  BOOL b = YPSC_GETWAVEPARA_PRIM(para)
#define YPSC_GETWAVEPARA_PRIM(para) \
(((para) & YP_SCWAVEPARA_PRIM) >> 29)
// 设置参数para的PRIM值为bPrim
#define YPSC_SETWAVEPARA_PRIM(para, bPrim) \
if (bPrim) {(para) = (para) | YP_SCWAVEPARA_PRIM; }\
else {(para) = (para) & (~YP_SCWAVEPARA_PRIM);}
// 获取SEC值如  BOOL b = YPSC_GETWAVEPARA_SEC(para)
#define YPSC_GETWAVEPARA_SEC(para) \
(((para) & YP_SCWAVEPARA_SEC) >> 28)
// 设置参数para的PRIM值为bPrim
#define YPSC_SETWAVEPARA_SEC(para, bSec) \
if (bSec) {(para) = (para) | YP_SCWAVEPARA_SEC; }\
else (para) = (para) & (~YP_SCWAVEPARA_SEC);
